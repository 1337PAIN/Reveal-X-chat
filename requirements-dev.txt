# ONNX export toolchain (milestone M2).
#
# These are NOT runtime dependencies. The server still classifies with the
# scikit-learn model; these are needed only to regenerate
# app/static/models/tamper_detector.onnx via
#   python scripts/export_tamper_detector_onnx.py
# and to run tests/test_onnx_export.py, which skips itself when onnxruntime is
# absent. The browser gets its runtime from app/static/vendor/onnxruntime/,
# which is vendored in the repo -- no npm, no CDN, works offline.
#
# Note the version gap: this onnxruntime validates the export server-side,
# while the browser runs the vendored onnxruntime-web 1.19.2. The export's
# opset has to stay within what 1.19.2 supports, so bumping these without
# also refreshing the vendored bundle is how in-browser detection breaks
# silently.
skl2onnx==1.20.0
onnx==1.23.2
onnxruntime==1.31.1
